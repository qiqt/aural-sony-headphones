cmake_minimum_required(VERSION 3.21)
project(AuralEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aural_engine BluetoothWrapper.cpp CommandSerializer.cpp Headphones.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Listener.cpp")
 target_sources(aural_engine PRIVATE Listener.cpp)
endif()
if(WIN32)
 target_sources(aural_engine PRIVATE windows/WindowsBluetoothConnector.cpp)
 target_include_directories(aural_engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(aural_engine PRIVATE ws2_32 Bthprops Rpcrt4 Setupapi)
 target_compile_definitions(aural_engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
